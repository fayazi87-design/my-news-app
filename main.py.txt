import os
from flask import Flask, request, jsonify
import yfinance as yf
import google.generativeai as genai

app = Flask(__name__)

# وصل شدن به مغز گوگل
genai.configure(api_key=os.environ.get("GEMINI_API_KEY"))
model = genai.GenerativeModel('gemini-1.5-flash')

def translate_text(text):
    prompt = f"ترجمه تخصصی بازار سرمایه به فارسی (نام شرکت‌ها انگلیسی بماند و قبلش کلمه شرکت بیاید): {text}"
    response = model.generate_content(prompt)
    return response.text

@app.route("/")
def home():
    # گرفتن تیکر از آدرس (مثلاً AAPL)
    symbol = request.args.get("ticker", "META").upper()
    stock = yf.Ticker(symbol)
    news = stock.news[:5] # ۵ خبر آخر
    
    results = []
    for item in news:
        results.append({
            "title_en": item['title'],
            "title_fa": translate_text(item['title']),
            "link": item['link']
        })
    return jsonify(results)

if __name__ == "__main__":
    app.run(host='0.0.0.0', port=int(os.environ.get("PORT", 5000)))
